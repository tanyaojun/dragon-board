[数据源层]
    ├── 东方财富API (websocket.ts) ──→ 实时行情
    ├── 八平台数据 (dataLoader.ts) ──→ 热榜排名
    └── KPL题材数据 (window.KPL_THEME_DATA) ──→ 题材列表

        ↓

[数据处理层]
    ├── dataLoader.ts 
    │   ├── loadStockDetails()  // 加载东财行情
    │   ├── loadAllPlatforms()   // 加载八平台排名
    │   └── mergeData()          // 合并数据 → window.allData.merged

    ↓

[缓存层] ←──────── LRUCache.ts
    ├── stockCache    // 股票数据缓存
    ├── leaderCache   // 龙头数据缓存  
    ├── sectorCache   // 题材数据缓存
    └── quoteCache    // 行情数据缓存

    ↓

[分析层]
    ├── sectorAnalyzer.ts
    │   ├── 从 window.KPL_THEME_DATA 加载题材列表
    │   ├── 从 API 加载题材详情
    │   ├── 建立 stockThemeMap / themeStockMap
    │   └── 计算热度、轮动、预警

    ├── DragonAnalyzer.ts
    │   ├── 依赖 algorithmManager 计算分数
    │   ├── 依赖 dragonBreathAnalyzer 获取情绪
    │   ├── 依赖 sectorAnalyzer 获取题材因子
    │   └── 输出 leaders / leaderChanges

    ├── DragonBreathAnalyzer.ts
    │   ├── 计算市场情绪 (涨停数、跌停数、炸板率等)
    │   └── 输出 phase (冰点/启动/发酵/高潮/退潮)

    └── AlgorithmManager.ts
        ├── 从 FACTORS 加载因子定义
        ├── 从 ALGORITHMS 加载算法配置
        ├── 依赖 sectorAnalyzer 获取题材因子
        ├── 依赖 dragonBreathAnalyzer 获取情绪因子
        └── 输出 calculateScore()

    ↓

[增量更新] ←─── IncrementalUpdater.ts
    ├── 监听 quotes-updated / dragon-updated 等事件
    ├── 按优先级将更新加入队列
    └── 批量处理 → 触发渲染

    ↓

[Store层] ←─── Pinia (stock.ts, theme.ts, etc.)
    ├── stockStore.stocks  // 响应式数据源
    └── 组件通过 storeToRefs 订阅

    ↓

[UI层]
    ├── App.vue (启动界面 → 主界面)
    ├── DataTable.vue (显示股票列表)
    └── 各面板组件 (SectorPanel/DragonHeadPanel等)
